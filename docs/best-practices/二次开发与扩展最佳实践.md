# 二次开发与扩展最佳实践

## 概述

llama.cpp 提供了丰富的 API 和扩展机制，支持开发者进行二次开发和功能扩展。本文档提供了 API 使用、扩展开发方法和最佳实践的详细指南。

## 核心 API 使用

### 1. C API

#### 基础使用

```c
#include "llama.h"

int main() {
    // 初始化后端
    llama_backend_init();
    
    // 模型参数
    llama_model_params model_params = llama_model_default_params();
    model_params.use_mlock = false;
    
    // 加载模型
    llama_model * model = llama_model_load_from_file("model.gguf", model_params);
    if (!model) {
        fprintf(stderr, "Failed to load model\n");
        return 1;
    }
    
    // 上下文参数
    llama_context_params ctx_params = llama_context_default_params();
    ctx_params.n_ctx = 2048;
    ctx_params.seed = 42;
    
    // 创建上下文
    llama_context * ctx = llama_init_from_model(model, ctx_params);
    if (!ctx) {
        fprintf(stderr, "Failed to create context\n");
        llama_model_free(model);
        return 1;
    }
    
    // 释放资源
    llama_free(ctx);
    llama_model_free(model);
    llama_backend_free();
    
    return 0;
}
```

#### 文本生成

```c
#include "llama.h"
#include "common.h"

int main() {
    llama_backend_init();
    
    llama_model_params model_params = llama_model_default_params();
    llama_model * model = llama_model_load_from_file("model.gguf", model_params);
    
    llama_context_params ctx_params = llama_context_default_params();
    ctx_params.n_ctx = 2048;
    llama_context * ctx = llama_init_from_model(model, ctx_params);
    
    // 准备输入
    const char * prompt = "The quick brown fox jumps over the lazy dog.";
    std::vector<llama_token> tokens = llama_tokenize(ctx, prompt, false);
    
    // 添加到批次
    llama_batch batch = llama_batch_init(tokens.size(), 0, 1);
    for (size_t i = 0; i < tokens.size(); i++) {
        batch.token[i] = tokens[i];
        batch.pos[i] = i;
        batch.n_tokens = i + 1;
        batch.logits[i] = false;
    }
    
    // 解码提示词
    llama_decode(ctx, batch);
    
    // 生成新 token
    int n_predict = 100;
    for (int i = 0; i < n_predict; i++) {
        // 获取下一个 token
        llama_token new_token = llama_sample_token_greedy(ctx, NULL);
        
        // 添加到批次
        llama_batch_clear(&batch);
        batch.token[0] = new_token;
        batch.pos[0] = tokens.size() + i;
        batch.n_tokens = 1;
        batch.logits[0] = true;
        
        // 解码
        llama_decode(ctx, batch);
        
        // 转换为文本
        std::string text = llama_token_to_piece(ctx, new_token);
        printf("%s", text.c_str());
        fflush(stdout);
    }
    
    printf("\n");
    
    // 清理
    llama_batch_free(batch);
    llama_free(ctx);
    llama_model_free(model);
    llama_backend_free();
    
    return 0;
}
```

## 扩展开发

### 1. 自定义采样器

```c
#include "llama.h"

// 自定义采样器实现
typedef struct {
    float temperature;
    float top_p;
    int top_k;
} custom_sampler_params;

llama_token custom_sample_token(
    llama_context * ctx,
    custom_sampler_params * params
) {
    // 获取 logits
    const float * logits = llama_get_logits(ctx);
    
    // 应用温度
    float scale = 1.0f / params->temperature;
    std::vector<float> scaled_logits;
    scaled_logits.resize(llama_n_vocab(ctx));
    for (int i = 0; i < llama_n_vocab(ctx); i++) {
        scaled_logits[i] = logits[i] * scale;
    }
    
    // 应用 Top-P
    if (params->top_p < 1.0f) {
        std::vector<std::pair<float, llama_token>> candidates;
        for (int i = 0; i < llama_n_vocab(ctx); i++) {
            candidates.emplace_back(scaled_logits[i], i);
        }
        
        std::sort(candidates.begin(), candidates.end(),
            [](const auto & a, const auto & b) { return a.first > b.first; });
        
        // 计算累积概率
        float cumulative_prob = 0.0f;
        std::vector<llama_token> filtered_tokens;
        for (const auto & candidate : candidates) {
            cumulative_prob += exp(candidate.first);
            filtered_tokens.push_back(candidate.second);
            if (cumulative_prob >= params->top_p) {
                break;
            }
        }
        
        // 从过滤后的 token 中采样
        int random_index = rand() % filtered_tokens.size();
        return filtered_tokens[random_index];
    }
    
    // 默认贪婪采样
    llama_token best_token = 0;
    float max_logit = -INFINITY;
    for (int i = 0; i < llama_n_vocab(ctx); i++) {
        if (scaled_logits[i] > max_logit) {
            max_logit = scaled_logits[i];
            best_token = i;
        }
    }
    return best_token;
}
```

```python
import llama_cpp
import logging

# 设置日志
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class SafeLlamaWrapper:
    def __init__(self, model_path: str, **kwargs):
        self.model_path = model_path
        self.kwargs = kwargs
        self.model = None
        
    def load_model(self):
        try:
            self.model = llama_cpp.Llama(
                model_path=self.model_path,
                **self.kwargs
            )
            logger.info(f"Model loaded successfully from {self.model_path}")
            return True
        except Exception as e:
            logger.error(f"Failed to load model: {e}")
            return False
    
    def generate(self, prompt: str, **kwargs):
        if not self.model:
            if not self.load_model():
                raise RuntimeError("Model not loaded")
        
        try:
            response = self.model(prompt, **kwargs)
            return response
        except Exception as e:
            logger.error(f"Generation failed: {e}")
            raise

# 使用示例
wrapper = SafeLlamaWrapper("model.gguf", n_ctx=2048)
if wrapper.load_model():
    response = wrapper.generate("Hello, world!", max_tokens=100)
```

#### 并发处理

```python
import concurrent.futures
import llama_cpp

# 模型池管理
class LlamaModelPool:
    def __init__(self, model_path: str, pool_size: int = 4):
        self.model_path = model_path
        self.pool_size = pool_size
        self.models = []
        
        for _ in range(pool_size):
            model = llama_cpp.Llama(
                model_path=model_path,
                n_ctx=2048,
                n_gpu_layers=-1,
                verbose=False
            )
            self.models.append(model)
    
    def get_model(self):
        # 简单轮询分配
        model = self.models[0]
        self.models.pop(0)
        self.models.append(model)
        return model

# 并发处理
def process_prompt(prompt: str, model_pool: LlamaModelPool):
    model = model_pool.get_model()
    try:
        response = model(prompt, max_tokens=100)
        return response['choices'][0]['text']
    finally:
        pass  # 模型自动返回到池中

# 批量处理
def batch_process(prompts: list[str]):
    model_pool = LlamaModelPool("model.gguf", pool_size=4)
    
    with concurrent.futures.ThreadPoolExecutor(max_workers=4) as executor:
        futures = []
        for prompt in prompts:
            future = executor.submit(process_prompt, prompt, model_pool)
            futures.append(future)
        
        results = []
        for future in concurrent.futures.as_completed(futures):
            try:
                result = future.result()
                results.append(result)
            except Exception as e:
                print(f"Error: {e}")
    
    return results
```

```c
// 模块化采样器设计
typedef struct {
    const char * name;
    void * params;
    llama_token (*sample_func)(llama_context *, void *);
} custom_sampler;

// 采样器注册表
std::vector<custom_sampler> samplers;

void register_sampler(const char * name, void * params, llama_token (*sample_func)(llama_context *, void *)) {
    samplers.push_back({name, params, sample_func});
}

llama_token use_sampler(const char * name, llama_context * ctx) {
    for (const auto & sampler : samplers) {
        if (strcmp(sampler.name, name) == 0) {
            return sampler.sample_func(ctx, sampler.params);
        }
    }
    return llama_sample_token_greedy(ctx, NULL);
}

// 使用示例
```

```python
import json
import yaml
from typing import Dict, Any
import llama_cpp

class LlamaConfig:
    def __init__(self, config_file: str):
        self.config = self.load_config(config_file)
    
    def load_config(self, config_file: str) -> Dict[str, Any]:
        if config_file.endswith('.json'):
            with open(config_file, 'r') as f:
                return json.load(f)
        elif config_file.endswith('.yaml') or config_file.endswith('.yml'):
            with open(config_file, 'r') as f:
                return yaml.safe_load(f)
        else:
            raise ValueError("Unsupported config format")
    
    def get_model_params(self) -> Dict[str, Any]:
        return self.config.get('model', {})
    
    def get_inference_params(self) -> Dict[str, Any]:
        return self.config.get('inference', {})
    
    def get_sampling_params(self) -> Dict[str, Any]:
        return self.config.get('sampling', {})
```

### 2. 性能分析

```python
import time
import llama_cpp

class PerformanceProfiler:
    def __init__(self, model: llama_cpp.Llama):
        self.model = model
        self.timings = {}
    
    def profile_generation(self, prompt: str, **kwargs):
        start_time = time.time()
        
        response = self.model(prompt, **kwargs)
        
        end_time = time.time()
        elapsed = end_time - start_time
        
        # 记录性能数据
        self.timings['elapsed_time'] = elapsed
        self.timings['tokens_generated'] = len(response['choices'][0]['text'].split())
        self.timings['tokens_per_second'] = self.timings['tokens_generated'] / elapsed
        
        return response
    
    def get_performance_stats(self) -> dict:
        return self.timings

# 使用示例
model = llama_cpp.Llama(model_path="model.gguf", verbose=False)
profiler = PerformanceProfiler(model)

response = profiler.profile_generation(
    "The quick brown fox jumps over the lazy dog.",
    max_tokens=100,
    temperature=0.8
)

stats = profiler.get_performance_stats()
print(f"Generation time: {stats['elapsed_time']:.2f}s")
print(f"Tokens generated: {stats['tokens_generated']}")
print(f"Tokens per second: {stats['tokens_per_second']:.2f}")
```
