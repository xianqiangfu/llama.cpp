#include "ggml.h"

void test_quantization() {
    // 准备测试数据
    const int n = 1000;
    std::vector<float> src_data(n);
    for (int i = 0; i < n; i++) {
        src_data[i] = (float)i / n;
    }
    
    // 测试不同量化类型
    const ggml_type quant_types[] = {
        GGML_TYPE_Q4_0, GGML_TYPE_Q4_1, GGML_TYPE_Q5_0,
        GGML_TYPE_Q5_1, GGML_TYPE_Q8_0, GGML_TYPE_F16
    };
    
    for (ggml_type type : quant_types) {
        const ggml_type_traits * qfns = ggml_get_type_traits(type);
        
        // 量化
        std::vector<uint8_t> quantized(ggml_type_size(type) * (n / ggml_blck_size(type)));
        qfns->from_float(src_data.data(), quantized.data(), n);
        
        // 反量化
        std::vector<float> dst_data(n);
        qfns->to_float(quantized.data(), dst_data.data(), n);
        
        // 计算误差
        double total_error = 0.0;
        for (int i = 0; i < n; i++) {
            double diff = src_data[i] - dst_data[i];
            total_error += diff * diff;
        }
        double rmse = sqrt(total_error / n);
        
        printf("Type %s: RMSE = %.6f\n", ggml_type_name(type), rmse);
        EXPECT_LT(rmse, 0.1);  // 允许一定的误差
    }
}

void test_end_to_end_inference() {
    llama_backend_init();
    
    // 加载模型
    llama_model_params model_params = llama_model_default_params();
    llama_model * model = llama_model_load_from_file("test_model.gguf", model_params);
    ASSERT_NE(model, nullptr);
    
    // 创建上下文
    llama_context_params ctx_params = llama_context_default_params();
    ctx_params.n_ctx = 1024;
    llama_context * ctx = llama_init_from_model(model, ctx_params);
    ASSERT_NE(ctx, nullptr);
    
    // 准备输入
    const char * prompt = "The capital of France is";
    std::vector<llama_token> tokens = llama_tokenize(ctx, prompt, false);
    
    // 创建批次
    llama_batch batch = llama_batch_init(tokens.size(), 0, 1);
    for (size_t i = 0; i < tokens.size(); i++) {
        batch.token[i] = tokens[i];
        batch.pos[i] = i;
        batch.n_tokens = i + 1;
        batch.logits[i] = false;
    }
    
    // 解码提示词
    int result = llama_decode(ctx, batch);
    ASSERT_EQ(result, 0);
    
    // 生成新 token
    std::string generated_text = prompt;
    for (int i = 0; i < 50; i++) {
        // 采样下一个 token
        llama_token new_token = llama_sample_token_greedy(ctx, NULL);
        
        // 转换为文本
        std::string token_text = llama_token_to_piece(ctx, new_token);
        generated_text += token_text;
        
        // 检查停止条件
        if (token_text == "\n" || token_text == ".") {
            break;
        }
        
        // 准备下一个批次
        llama_batch_clear(&batch);
        batch.token[0] = new_token;
        batch.pos[0] = tokens.size() + i;
        batch.n_tokens = 1;
        batch.logits[0] = true;
        
        // 解码
        result = llama_decode(ctx, batch);
        ASSERT_EQ(result, 0);
    }
    
    // 验证输出
    ASSERT_FALSE(generated_text.empty());
    ASSERT_GT(generated_text.length(), prompt.length());
    
    printf("Generated text: %s\n", generated_text.c_str());
    
    // 清理
    llama_batch_free(batch);
    llama_free(ctx);
    llama_model_free(model);
    llama_backend_free();
    
    printf("✓ End-to-end inference test passed\n");
}

void test_performance() {
    llama_backend_init();

    llama_model * model = llama_model_load_from_file("test_model.gguf", model_params);

    ctx_params.n_ctx = 1024;
    llama_context * ctx = llama_init_from_model(model, ctx_params);
    
    // 准备测试输入
    const char * prompt = "The quick brown fox jumps over the lazy dog.";
    auto tokens = llama_tokenize(ctx, prompt, false);
    
    llama_batch batch = llama_batch_init(tokens.size(), 0, 1);
    for (size_t i = 0; i < tokens.size(); i++) {
        batch.token[i] = tokens[i];
        batch.pos[i] = i;
        batch.n_tokens = i + 1;
        batch.logits[i] = false;
    }
    
    // 测量解码时间
    auto start = std::chrono::high_resolution_clock::now();
    
    int result = llama_decode(ctx, batch);
    auto end = std::chrono::high_resolution_clock::now();
    
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);
    
    ASSERT_EQ(result, 0);
    
    printf("Decoding time: %ld ms\n", duration.count());
    printf("Tokens per second: %.2f\n", tokens.size() * 1000.0 / duration.count());
    
    llama_batch_free(batch);
    llama_free(ctx);
    llama_model_free(model);
    llama_backend_free();
    
    printf("✓ Performance test passed\n");
}

#!/bin/bash
# validate_quality.sh

MODEL=$1
TEST_FILE=${2:-"test.txt"}

echo "Validating model quality for: $MODEL"

# 计算困惑度
PERPLEXITY=$(./llama-perplexity -m $MODEL -f $TEST_FILE | grep -oP 'Final: \K[0-9.]+')
echo "Perplexity: $PERPLEXITY"

# 评估困惑度
if (( $(echo "$PERPLEXITY < 10.0" | bc -l) )); then
    echo "✓ Excellent quality (perplexity < 10.0)"
elif (( $(echo "$PERPLEXITY < 20.0" | bc -l) )); then
    echo "✓ Good quality (perplexity < 20.0)"
elif (( $(echo "$PERPLEXITY < 50.0" | bc -l) )); then
    echo "⚠ Acceptable quality (perplexity < 50.0)"
else
    echo "✗ Poor quality (perplexity >= 50.0)"
    exit 1
fi
