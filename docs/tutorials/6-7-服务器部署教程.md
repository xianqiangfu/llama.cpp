## 服务器概述

### llama-server 功能

llama-server 提供以下功能：

- **HTTP API 服务**：RESTful API 接口
- **OpenAI 兼容**：与 OpenAI API 兼容的接口
- **Web UI**：内置 Web 界面
- **多用户支持**：并发请求处理
- **流式响应**：Server-Sent Events 支持
- **多种推理模式**：完成、聊天、嵌入等
- **模版系统**：Jinja2 模板支持

### 服务器架构

```
llama-server
├── HTTP 服务器 (cpp-httplib)
├── 路由系统
├── 推理引擎
├── 批处理管理器
├── KV 缓存管理器
└── 流式输出处理器
```

## 生产部署

### Systemd 服务

```bash
# 1. 创建服务文件
sudo nano /etc/systemd/system/llama-server.service

# 2. 服务配置
[Unit]
Description=LLaMA.cpp HTTP Server
After=network.target

[Service]
Type=simple
User=llama
Group=llama
WorkingDirectory=/opt/llama.cpp
Environment="LLAMA_ARG_MODEL=/opt/llama.cpp/models/gemma-3-1b-it-Q4_K_M.gguf"
ExecStart=/opt/llama.cpp/build/bin/llama-server \
    -m /opt/llama.cpp/models/gemma-3-1b-it-Q4_K_M.gguf \
    --host 0.0.0.0 \
    --port 8080 \
    -c 8192 \
    -t 8 \
    -np 4 \
    --timeout 300

Restart=always
RestartSec=10
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target

# 3. 启动服务
sudo systemctl daemon-reload
sudo systemctl enable llama-server
sudo systemctl start llama-server

# 4. 检查状态
sudo systemctl status llama-server
sudo journalctl -u llama-server -f
```

### Nginx 反向代理

```bash
# 1. 安装 Nginx
sudo apt install nginx  # Ubuntu/Debian
sudo yum install nginx  # CentOS/RHEL

# 2. 配置反向代理
sudo nano /etc/nginx/sites-available/llama-server

# 3. 代理配置
upstream llama_backend {
    server 127.0.0.1:8080;
    keepalive 32;
}

server {
    listen 80;
    server_name api.example.com;

    # 基础代理设置
    location / {
        proxy_pass http://llama_backend;
        proxy_http_version 1.1;
        
        # 超时设置
        proxy_connect_timeout 300;
        proxy_send_timeout 300;
        proxy_read_timeout 300;
        
        # 头信息
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # 支持 SSE
        proxy_buffering off;
        proxy_cache off;
        proxy_set_header Connection '';
        chunked_transfer_encoding off;
    }
    
    # 静态文件缓存
    location ~* \.(js|css|png|jpg|jpeg|gif|ico)$ {
        proxy_pass http://llama_backend;
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
    
    # 请求体大小限制
    client_max_body_size 10M;
}

# 4. 启用站点
sudo ln -s /etc/nginx/sites-available/llama-server /etc/nginx/sites-enabled/
sudo nginx -t
sudo systemctl reload nginx
```

### 健康检查

```bash
#!/bin/bash
# health-check.sh

# 基础健康检查
check_health() {
    response=$(curl -s http://localhost:8080/health)
    
    if [ "$response" == '{"status":"ok"}' ]; then
        echo "服务器健康"
        return 0
    else
        echo "服务器不健康: $response"
        return 1
    fi
}

# 详细检查
check_details() {
    echo "=== 服务器状态 ==="
    curl -s http://localhost:8080/props | jq '.'
    
    echo "=== 活跃插槽 ==="
    curl -s http://localhost:8080/slots | jq '.'
}

# 性能检查
check_performance() {
    start_time=$(date +%s)
    
    curl -s http://localhost:8080/completion \
        -H "Content-Type: application/json" \
        -d '{"prompt": "测试", "n_predict": 10}' > /dev/null
    
    end_time=$(date +%s)
    duration=$((end_time - start_time))
    
    echo "响应时间: ${duration}s"
}

# 主循环
while true; do
    check_health
    check_performance
    sleep 60
done
```

# failover.sh

instances=(
    "http://localhost:8081"
    "http://localhost:8082"
    "http://localhost:8083"

get_healthy_instance() {
    for instance in "${instances[@]}"; do
        if curl -sf "$instance/health" > /dev/null; then

        -d '{"prompt": "测试"}'

### 自动扩缩容

```bash
#!/bin/bash
# auto-scale.sh

MIN_INSTANCES=2
MAX_INSTANCES=8
CPU_THRESHOLD=80
MEMORY_THRESHOLD=80

get_instance_count() {
    pgrep -f "llama-server" | wc -l
}

get_cpu_usage() {
    top -bn1 | grep "Cpu(s)" | awk '{print $2}' | cut -d'%' -f1
}

get_memory_usage() {
    free | grep Mem | awk '{print $3/$2 * 100.0}'
}

scale_up() {
    current_count=$(get_instance_count)
    if [ $current_count -lt $MAX_INSTANCES ]; then
        echo "扩容: 启动新实例"
        llama-server \
            -m "models/gemma-3-1b-it-Q4_K_M.gguf" \
            --port $((8080 + $current_count + 1)) &
    fi
}

scale_down() {
    current_count=$(get_instance_count)
    if [ $current_count -gt $MIN_INSTANCES ]; then
        echo "缩容: 停止实例"
        pkill -f "llama-server.*--port.*$(($current_count + 8080))"
    fi
}

# 主循环
while true; do
    cpu_usage=$(get_cpu_usage)
    memory_usage=$(get_memory_usage)
    
    echo "CPU: ${cpu_usage}%, Memory: ${memory_usage}%"
    
    if (( $(echo "$cpu_usage > $CPU_THRESHOLD" | bc -l) )); then
        scale_up
    elif (( $(echo "$cpu_usage < 50" | bc -l) )); then
        scale_down
    fi
    
    sleep 60
done
```
