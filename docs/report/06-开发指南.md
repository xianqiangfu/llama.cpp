# 开发指南

## 概述

本章面向开发者，详细介绍 llama.cpp 的开发环境搭建、API 使用、扩展开发和最佳实践。

## 1. 开发环境搭建

### 1.1 开发工具

- **编译器**: GCC 7+、Clang 6+、MSVC 2019+
- **构建工具**: CMake 3.14+、Make/Ninja
- **版本控制**: Git
- **调试器**: GDB/LLDB 或 Visual Studio Debugger

#### 推荐工具

- **IDE**: CLion、Visual Studio、VSCode
- **代码格式化**: clang-format
- **静态分析**: clang-tidy
- **性能分析**: perf、VTune

## 2. API 使用

### 2.1 C API 基础

#### 模型加载

```c
#include "llama.h"

int main() {
    // 设置模型参数
    llama_model_params model_params = llama_model_default_params();
    model_params.n_gpu_layers = 35;  // GPU 层数
    
    // 加载模型
    llama_model * model = llama_load_model_from_file(
        "model.gguf", model_params);
    
    if (!model) {
        fprintf(stderr, "Failed to load model\n");
        return 1;
    }
    
    // 使用模型...
    
    // 释放模型
    llama_free_model(model);
    
    return 0;
}
```

### 2.3 批次处理

```c
// 创建批次
llama_batch batch = llama_batch_init(n_tokens, 0, 1);

// 填充批次
for (int i = 0; i < n_tokens; i++) {
    batch.token[i] = tokens[i];
    batch.pos[i] = i;
    batch.n_seq_id[i] = 1;
    batch.seq_id[i][0] = 0;
    batch.logits[i] = (i == n_tokens - 1);  // 最后一个 token 输出 logits

batch.n_tokens = n_tokens;

// 执行解码
if (llama_decode(ctx, batch) != 0) {
    fprintf(stderr, "Failed to decode\n");
    llama_batch_free(batch);
    return 1;
}

// 获取 logits
float * logits = llama_get_logits_ith(ctx, -1);

// 释放批次

// 在 llama-model.cpp 中实现模型特定逻辑
static void llm_model_my_model_init(struct ggml_context * ctx,
                                    struct my_model_params params) {
    // 初始化模型权重
    model->tok_embd = ggml_new_tensor_2d(ctx, GGML_TYPE_F32, 
                                         params.n_embd, params.n_vocab);
    model->layers.resize(params.n_layer);
    
    for (int i = 0; i < params.n_layer; i++) {
        struct llama_layer * layer = &model->layers[i];
        layer->attn_norm = ggml_new_tensor_1d(ctx, GGML_TYPE_F32, params.n_embd);
        layer->wq = ggml_new_tensor_2d(ctx, GGML_TYPE_Q4_K, 
                                       params.n_embd, params.n_embd);
        // ... 其他层
    }
}

static void llm_model_my_model_forward(struct ggml_context * ctx,
                                       struct my_model_state state,
                                       struct ggml_cgraph * gf) {
    // 前向传播逻辑
    ggml_tensor * inpL = state.embd;
    
    for (int il = 0; il < model->hparams.n_layer; il++) {
        struct llama_layer * layer = &model->layers[il];
        
        // 自注意力
        ggml_tensor * cur = llama_build_attn(ctx, state, gf, layer, il, inpL);
        
        // 前馈网络
        cur = llama_build_ffn(ctx, state, gf, layer, il, cur);
        
        // 残差连接
        inpL = ggml_add(ctx, inpL, cur);
    }
    
    state.output = inpL;
}
```

### 3.2 新采样策略

#### 实现自定义采样器

```cpp
// 自定义采样器实现
struct my_sampler_context {
    float temperature;
    int top_k;
    float top_p;
};

static llama_token_data * my_sampler_apply(
    void * ctx_data, llama_token_data_array * candidates) {
    struct my_sampler_context * ctx = ctx_data;
    
    // 1. 应用温度
    for (size_t i = 0; i < candidates->size; i++) {
        candidates->data[i].logit /= ctx->temperature;
    }
    
    // 2. 应用 Top-K
    llama_token_data_array_partial_sort_inplace(candidates, ctx->top_k);
    candidates->size = ctx->top_k;
    
    // 3. 应用 Top-P
    llama_sampler_softmax_impl(candidates, false);
    
    float cum_sum = 0.0f;
    size_t last_idx = candidates->size;
    for (size_t i = 0; i < candidates->size; i++) {
        cum_sum += candidates->data[i].p;
        if (cum_sum >= ctx->top_p) {
            last_idx = i + 1;
            break;
        }
    }
    candidates->size = last_idx;
    
    return candidates;
}

static void my_sampler_reset(void * ctx_data) {
    // 重置采样器状态
}

static void my_sampler_free(void * ctx_data) {
    free(ctx_data);
}

// 注册采样器
struct llama_sampler_i llama_sampler_my_sampler_interface = {
    .name = "my_sampler",
    .apply = my_sampler_apply,
    .reset = my_sampler_reset,
    .free = my_sampler_free,
};

llama_sampler_t * llama_sampler_init_my_sampler(float temperature,
                                                  int top_k, float top_p) {
    struct my_sampler_context * ctx = malloc(sizeof(*ctx));
    ctx->temperature = temperature;
    ctx->top_k = top_k;
    ctx->top_p = top_p;
    
    return llama_sampler_init(&llama_sampler_my_sampler_interface, ctx);
}
```

### 3.4 新量化格式

#### 实现量化算法

```cpp
// 量化函数
static void quantize_row_my_format(const float * x, void * vy, int k) {
    assert(k % BLOCK_SIZE == 0);
    
    const int nb = k / BLOCK_SIZE;
    block_my_format * y = (block_my_format *)vy;
    
    for (int i = 0; i < nb; i++) {
        // 1. 找到最大值
        float amax = 0.0f;
        for (int j = 0; j < BLOCK_SIZE; j++) {
            float val = fabsf(x[i * BLOCK_SIZE + j]);
            if (val > amax) amax = val;
        }
        
        // 2. 计算缩放因子
        float d = amax / 7.0f;
        y[i].d = ggml_fp32_to_fp16(d);
        
        // 3. 量化
        float id = 1.0f / d;
        for (int j = 0; j < BLOCK_SIZE; j += 2) {
            float x0 = x[i * BLOCK_SIZE + j] * id + 8.5f;
            float x1 = x[i * BLOCK_SIZE + j + 1] * id + 8.5f;
            
            uint8_t v0 = (uint8_t)MIN(MAX(0, (int)x0), 15);
            uint8_t v1 = (uint8_t)MIN(MAX(0, (int)x1), 15);
            
            y[i].qs[j / 2] = (v0 << 4) | v1;
        }
    }
}

// 反量化函数
static void dequantize_row_my_format(const void * vy, float * x, int k) {
    assert(k % BLOCK_SIZE == 0);
    
    const int nb = k / BLOCK_SIZE;
    const block_my_format * y = (const block_my_format *)vy;
    
    for (int i = 0; i < nb; i++) {
        float d = ggml_fp16_to_fp32(y[i].d);
        float id = 7.0f * d;
        
        for (int j = 0; j < BLOCK_SIZE; j += 2) {
            uint8_t v0 = (y[i].qs[j / 2] >> 4) & 0xF;
            uint8_t v1 = y[i].qs[j / 2] & 0xF;
            
            x[i * BLOCK_SIZE + j] = ((float)v0 - 8.0f) * id;
            x[i * BLOCK_SIZE + j + 1] = ((float)v1 - 8.0f) * id;
        }
    }
}

// 注册量化格式
struct ggml_type_traits ggml_type_traits_my_format = {
    .type_name = "my_format",
    .blck_size = BLOCK_SIZE,
    .type_size = sizeof(block_my_format),
    .is_quantized = true,
    .to_float = dequantize_row_my_format,
    .from_float = quantize_row_my_format,
    .from_float_reference = quantize_row_my_format,
};
```

// 完善的错误处理
bool safe_load_model(const char * path, llama_model ** model_out) {
    // 1. 检查参数
    if (!path || !model_out) {
        fprintf(stderr, "Invalid parameters\n");
        return false;
    }
    
    // 2. 检查文件存在
    FILE * f = fopen(path, "rb");
    if (!f) {
        fprintf(stderr, "File not found: %s\n", path);
        return false;
    }
    fclose(f);
    
    // 3. 加载模型
    llama_model_params model_params = llama_model_default_params();
    llama_model * model = llama_load_model_from_file(path, model_params);
    if (!model) {
        fprintf(stderr, "Failed to load model from %s\n", path);
        return false;
    }
    
    // 4. 返回模型
    *model_out = model;
    return true;
}
